{{- /* Inlines the SVG that scripts/render-diagrams.mjs pre-rendered for one diagram fence. Every render-codeblock-.html hook calls this with its own context. The file is diagrams//.svg, key = sha256(page folder "\n" attribute "\n" .Inner) — the driver computes the same key; change one side and every lookup misses (errorf below). The attribute is the language's drawing attribute as written — `dialect` on spice, `direction` on onnx (ADR 0008) — and empty for every other language. */ -}} {{- $lang := .Type -}} {{- $folder := "" -}} {{- with .Page.File }}{{ $folder = .Dir }}{{ end -}} {{- $attribute := "" -}} {{- with index (dict "spice" "dialect" "onnx" "direction") $lang -}} {{- with index $.Attributes . }}{{ $attribute = string . }}{{ end -}} {{- end -}} {{- $key := sha256 (printf "%s\n%s\n%s" $folder $attribute .Inner) -}} {{- $align := "" -}} {{- with .Attributes.align -}} {{- if not (in (slice "left" "center" "right") .) -}} {{- errorf "%s: %s fence align=%q: use left, center or right" $.Position $lang . -}} {{- end -}} {{- $align = printf " diagram-align-%s" . -}} {{- end -}} {{- with resources.Get (printf "diagrams/%s/%s.svg" $lang $key) -}} {{- /* Natural width from the viewBox, so a small diagram never stretches to the column. Graphviz viewBox units are points; the others', onnx's ELK px included, are px. */ -}} {{- $width := 0 -}} {{- with findRESubmatch `]*\bviewBox="([^"]+)"` .Content 1 -}} {{- $width = index (split (replaceRE `[\s,]+` " " (trim (index (index . 0) 1) " ")) " ") 2 | float -}} {{- end -}} {{- if eq $lang "graphviz" }}{{ $width = mul $width (div 4.0 3.0) }}{{ end -}} {{- else -}} {{- errorf "%s: %s fence was not pre-rendered. Run: npm run diagrams" $.Position $lang -}} {{- end -}}
